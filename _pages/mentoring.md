<script>
document.addEventListener('DOMContentLoaded', function () {
  const carousel = document.querySelector('#review-carousel');
  if (!carousel) return;

  const slides = Array.from(carousel.querySelectorAll('.review-slide'));
  const prevBtn = document.querySelector('[data-carousel-prev]');
  const nextBtn = document.querySelector('[data-carousel-next]');
  const dotsContainer = document.querySelector('[data-carousel-dots]');
  let currentIndex = 0;

  const goToSlide = (targetIndex) => {
    slides[currentIndex].classList.remove('is-active');
    currentIndex = (targetIndex + slides.length) % slides.length;
    slides[currentIndex].classList.add('is-active');
    updateDots();
  };

  const createDot = (index) => {
    const dot = document.createElement('button');
    dot.type = 'button';
    dot.className = 'mentoring-reviews__dot' + (index === currentIndex ? ' is-active' : '');
    dot.setAttribute('aria-label', `Перейти к отзыву ${index + 1}`);
    dot.addEventListener('click', () => goToSlide(index));
    return dot;
  };

  const updateDots = () => {
    dotsContainer.innerHTML = '';
    slides.forEach((_, index) => dotsContainer.appendChild(createDot(index)));
  };

  prevBtn?.addEventListener('click', () => goToSlide(currentIndex - 1));
  nextBtn?.addEventListener('click', () => goToSlide(currentIndex + 1));

  updateDots();
});
</script>
